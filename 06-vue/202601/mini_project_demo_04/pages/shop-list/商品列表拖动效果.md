### 2、初始化删除按钮宽度
```
getEleWidth(w) {
    let real = 0;
    try {
      let res = wx.getSystemInfoSync().windowWidth
      console.log(res, 'res')
      // 窗口是给的px单位，需要换算成rpx
      let scale = (750 / 2) / (w / 2)
      real = Math.floor(res / scale);
      console.log(real, 'real');

      return real;
    } catch (e) {
      return false;
      // Do something when catch error
    }
}
```
